cmake_minimum_required(VERSION 3.16)
project(VirtualCell LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(virtual_cell INTERFACE)
target_include_directories(virtual_cell INTERFACE include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(virtual_cell INTERFACE -Wall -Wextra -Wpedantic)
endif()
add_executable(virtual_cell_demo src/main.cpp)
target_link_libraries(virtual_cell_demo PRIVATE virtual_cell)
enable_testing()
add_executable(virtual_cell_tests tests/tests.cpp)
target_link_libraries(virtual_cell_tests PRIVATE virtual_cell)
add_test(NAME virtual_cell_checks COMMAND virtual_cell_tests)
add_executable(chess_memory_demo src/chess_memory.cpp)
target_link_libraries(chess_memory_demo PRIVATE virtual_cell)
add_executable(memory_tests tests/memory_tests.cpp)
target_link_libraries(memory_tests PRIVATE virtual_cell)
add_test(NAME relational_memory_checks COMMAND memory_tests)
add_executable(objectives_demo src/objectives_demo.cpp)
target_link_libraries(objectives_demo PRIVATE virtual_cell)
add_executable(objectives_tests tests/objectives_tests.cpp)
target_link_libraries(objectives_tests PRIVATE virtual_cell)
add_test(NAME objective_summary_checks COMMAND objectives_tests)
add_executable(person_model_demo src/person_model_demo.cpp)
target_link_libraries(person_model_demo PRIVATE virtual_cell)
add_executable(person_model_tests tests/person_model_tests.cpp)
target_link_libraries(person_model_tests PRIVATE virtual_cell)
add_test(NAME integrated_model_checks COMMAND person_model_tests)

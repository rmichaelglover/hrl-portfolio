cmake_minimum_required(VERSION 3.16)
project(alex_observation)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
add_executable(observer_node src/observer_node.cpp)
target_compile_features(observer_node PUBLIC cxx_std_17)
target_include_directories(observer_node PRIVATE ../../include)
ament_target_dependencies(observer_node rclcpp geometry_msgs sensor_msgs diagnostic_msgs)
install(TARGETS observer_node DESTINATION lib/${PROJECT_NAME})
ament_package()

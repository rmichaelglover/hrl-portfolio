cmake_minimum_required(VERSION 3.16)
project(VirtualCell LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(virtual_cell INTERFACE)
target_include_directories(virtual_cell INTERFACE include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(virtual_cell INTERFACE -Wall -Wextra -Wpedantic)
endif()
add_executable(virtual_cell_demo src/main.cpp)
target_link_libraries(virtual_cell_demo PRIVATE virtual_cell)

cmake_minimum_required(VERSION 3.16)
project(MagnumChessicusCPP LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(magnum INTERFACE)
target_include_directories(magnum INTERFACE include)
target_compile_options(magnum INTERFACE -Wall -Wextra -Wpedantic)
add_executable(magnum_cpp src/main.cpp)
target_link_libraries(magnum_cpp PRIVATE magnum)
enable_testing()
add_executable(magnum_tests tests/tests.cpp)
target_link_libraries(magnum_tests PRIVATE magnum)
add_test(NAME magnum_sound_bounds COMMAND magnum_tests)
add_executable(magnum_benchmark src/benchmark.cpp)
target_link_libraries(magnum_benchmark PRIVATE magnum)
add_executable(magnum_verify src/verify.cpp)
target_link_libraries(magnum_verify PRIVATE magnum)
add_executable(magnum_compare src/compare.cpp)
